Firestore rules

The text to paste into the Firebase console, so the Medicine Timer, Renewals and Returns apps can save your data.
On an iPhone: tap Copy above, then open console.firebase.google.com in another tab, go to your project → Firestore → Rules, select everything in the box, delete it, paste, and tap Publish.

What you're pasting

rules_version = '2';

// Firestore rules for packing-list-3c532. The packing list is open on
// purpose; everything else is claimed by whoever signs in first and is
// private to them afterwards. To share one later, add an email to a
// "sharedWith" list on the document itself -- honoured below already.

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }
    // Unclaimed documents are fair game for the first person to sign in --
    // that is how each app takes ownership on first run.
    function unclaimed(d) {
      return !("ownerUid" in d) && !("ownerEmail" in d);
    }

    // Matching the email as well as the account id means the same person is
    // recognised whichever way they signed in. Requiring email_verified is
    // what stops someone typing your address into a new account to get in.
    function owns(d) {
      return d.get("ownerUid", "") == request.auth.uid
        || (request.auth.token.get("email_verified", false) == true
            && d.get("ownerEmail", "") != ""
            && d.get("ownerEmail", "") == request.auth.token.get("email", ""));
    }
    function sharedTo(d) {
      return request.auth.token.get("email_verified", false) == true
        && request.auth.token.get("email", "") in d.get("sharedWith", []);
    }
    function mayOpen() {
      return signedIn() && (resource == null
        || unclaimed(resource.data)
        || owns(resource.data)
        || sharedTo(resource.data));
    }

    // Claiming means putting your own account id on it, not someone else's.
    function claimingForSelf() {
      return signedIn()
        && request.resource.data.get("ownerUid", "") == request.auth.uid;
    }

    // Open on purpose: works on a phone handed to anyone in the family.
    match /packingLists/{listId} {
      allow read, write: if true;
    }

    match /subscriptionLedgers/{ledgerId} {
      allow read: if mayOpen();
      allow create: if claimingForSelf();
      allow update, delete: if mayOpen();
    }

    // Medicine Timer, Renewals and Returns.
    match /homeApps/{appId} {
      allow read: if mayOpen();
      allow create: if claimingForSelf();
      allow update, delete: if mayOpen();
    }

    // Anything not named above is closed.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
